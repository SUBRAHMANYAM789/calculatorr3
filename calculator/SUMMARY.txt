📖 Example Usage:
-----------------
// Basic math
calc.add(5, 3)              // 8
calc.multiply(6, 7)         // 42

// Advanced math
calc.squareRoot(144)        // 12
calc.factorial(5)           // 120
calc.power(2, 8)            // 256

// Statistics
calc.average([5,10,15,20])  // 12.5
calc.median([1,2,3,4,5])    // 3

// Trigonometry
calc.sine(Math.PI/2)        // 1
calc.degreesToRadians(90)   // 1.5707...

// Financial
calc.simpleInterest(1000, 5, 2)     // 100
calc.compoundInterest(1000, 5, 2)   // 102.5
